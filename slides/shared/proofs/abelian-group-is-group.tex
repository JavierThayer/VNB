\begin{proposition}[abelian-group-is-group]
Suppose $s \text{ s.t. } \operatorname{is\text{-}abelian\text{-}group}(s)$. Then
\begin{equation*}
\operatorname{is\text{-}group}(s)
\end{equation*}

\end{proposition}

\begin{proof}
\begin{itemize}
  \item[\textbf{3}] By $\operatorname{mac\text{-}h}$.
  \item[\textbf{4--27}] By $\operatorname{is\text{-}group}$, reduce to
\begin{equation*}
\begin{array}{@{}l@{}} (\operatorname{length}(s) = 4) \wedge \\ (\operatorname{carr}(s) \in \mathbf{Set}) \wedge \\ \operatorname{opr}(s) \\ \in\; ((\operatorname{carr}(s) \times \operatorname{carr}(s)) \to \operatorname{carr}(s)) \wedge \\ (\operatorname{iden}(s) \in \operatorname{carr}(s)) \wedge \\ (\operatorname{inv}(s) \in (\operatorname{carr}(s) \to \operatorname{carr}(s))) \wedge \\ \operatorname{is\text{-}associative}(\operatorname{opr}(s), \operatorname{carr}(s)) \wedge \\ \operatorname{is\text{-}identity}(\operatorname{opr}(s), \operatorname{iden}(s), \operatorname{carr}(s)) \wedge \\ \operatorname{has\text{-}inverses}(\operatorname{opr}(s), \operatorname{iden}(s), \operatorname{inv}(s), \operatorname{carr}(s)) \end{array}
\end{equation*}
  \emph{(holds by assumption)}
\end{itemize}
\end{proof}
