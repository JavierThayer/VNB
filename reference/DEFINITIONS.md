Defined by primitive recursion on `NN` (`def-by-nn-recursion`): a `name-zero` base and a `name-succ` step.  The extra parameters range over a class (a ring, a monoid, ...), so each is a **functoid**, not a function.

### comb-kk

    forall([r, x, y], comb-kk(r, x, y, 0) == vnb-lambda(k, zz, if(k = 0, one(r), zero(r))))

    forall([r, x, y, m in nn], comb-kk(r, x, y, succ(m)) == vnb-lambda(k, zz, (add(r))((mul(r))(x, (comb-kk(r, x, y, m))(k - 1)), (mul(r))(y, (comb-kk(r, x, y, m))(k)))))

### dc-iter

    forall([s, a, nxt], dc-iter(s, a, nxt, 0) == a)

    forall([s, a, nxt, n in nn], dc-iter(s, a, nxt, succ(n)) == choice({y_ in s: y_ in nxt(n, dc-iter(s, a, nxt, n))}))

### det

    forall([r, a], det(r, 0, a) == one(r))

    forall([r, n, a], is-ring(r) implies n in nn implies a in mat(succ(n), succ(n), carr(r)) implies det(r, succ(n), a) == finsum(ring-additive-ag(r), vnb-lambda(j, interval(1, succ(n)), (mul(r))(mpow(ring-multiplicative-monoid(r), (neg(r))(one(r)), succ(j)), (mul(r))(entry(a, 1, j), det(r, n, minor(a, 1, j, n))))), interval(1, succ(n))))

### factorial

    factorial(0) = succ(0)

    forall([k in nn], factorial(succ(k)) = succ(k) * factorial(k))

### falling

    forall([n], falling(n, 0) == succ(0))

    forall([n, k in nn], falling(n, succ(k)) == (n - k) * falling(n, k))

### mpow

    forall([m, x], mpow(m, x, 0) == iden(m))

    forall([m, x, n in nn], mpow(m, x, succ(n)) == (opr(m))(x, mpow(m, x, n)))

### nth-deriv

    forall([f], nth-deriv(f, 0) == f)

    forall([f, n in nn], nth-deriv(f, succ(n)) == vnb-lambda(x, rr, deriv(nth-deriv(f, n), x)))

### nth-deriv-on

    forall([ndvk_, ndvu_, ndvf_], nth-deriv-on(ndvk_, ndvu_, ndvf_, 0) == ndvf_)

    forall([ndvk_, ndvu_, ndvf_, ndvn_ in nn], nth-deriv-on(ndvk_, ndvu_, ndvf_, succ(ndvn_)) == vnb-lambda(ndvz_, ndvu_, deriv-on(ndvk_, ndvu_, nth-deriv-on(ndvk_, ndvu_, ndvf_, ndvn_), ndvz_)))

### nth-deriv-v

    forall([m, f], nth-deriv-v(m, f, 0) == f)

    forall([m, f, n in nn], nth-deriv-v(m, f, succ(n)) == vnb-lambda(x, rr, deriv-v(m, nth-deriv-v(m, f, n), x)))

### prod-ord

    forall([m, f], prod-ord(m, f, 0) == iden(m))

    forall([m, f, n in nn], prod-ord(m, f, succ(n)) == (opr(m))(prod-ord(m, f, n), f(n)))

### reduce

    forall([op, f], reduce(op, f, 1) == f(0))

    forall([op, f, n], n in nn and 1 <= n implies reduce(op, f, succ(n)) == op(reduce(op, f, n), f(n)))

### ring-prod-n

    forall([f], ring-prod-n(f, 0) == zero-ring)

    forall([f, n in nn], ring-prod-n(f, succ(n)) == ring-prod(ring-prod-n(f, n), f(n)))

### sum

    forall([r, f], sum(r, f, 0) == zero(r))

    forall([r, f, n in nn], sum(r, f, succ(n)) == (add(r))(sum(r, f, n), f(n)))

### sum-ag

    forall([ag, f], sum-ag(ag, f, 0) == iden(ag))

    forall([ag, f, n in nn], sum-ag(ag, f, succ(n)) == (opr(ag))(sum-ag(ag, f, n), f(n)))

### taylor-poly-v

    forall([m, f, a, x], taylor-poly-v(m, f, a, x, 0) == (act(m))((x - a) ^ 0 * recip(factorial(0)), (nth-deriv-v(m, f, 0))(a)))

    forall([m, f, a, x, n in nn], taylor-poly-v(m, f, a, x, succ(n)) == (vadd(m))(taylor-poly-v(m, f, a, x, n), (act(m))((x - a) ^ succ(n) * recip(factorial(succ(n))), (nth-deriv-v(m, f, succ(n)))(a))))

### trinum

    trinum(0) == 0

    forall([n_ in nn], trinum(succ(n_)) == trinum(n_) + succ(n_))

### zen

    forall([x], zen(x, 0) = choice(x))

    forall([x, alpha in ord], zen(x, succ_ord(alpha)) = choice({y_ in x: forall([c_], ord-le(c_, alpha) implies not(zen(x, c_) = y_))}))

    forall([x, lam], limit-ord(lam) implies zen(x, lam) = choice({y_ in x: forall([c_], ord-lt(c_, lam) implies not(zen(x, c_) = y_))}))

### zkept

    forall([phi, grd, porel], zkept(phi, grd, porel, 0) = empty-set)

    forall([phi, grd, porel, alpha in ord], zkept(phi, grd, porel, succ_ord(alpha)) = union(zkept(phi, grd, porel, alpha), keep-set(phi, grd, porel, zkept(phi, grd, porel, alpha), alpha)))

    forall([phi, grd, porel, lam], limit-ord(lam) implies zkept(phi, grd, porel, lam) = big-union(beta, ord-segment(lam), zkept(phi, grd, porel, beta)))

### zup

    forall([grd, porel], zup(grd, porel, 0) = choice(grd))

    forall([grd, porel, alpha in ord], zup(grd, porel, succ_ord(alpha)) = choice({y_ in grd: is-strictly-below(porel, zup(grd, porel, alpha), y_)}))

    forall([grd, porel, lam], limit-ord(lam) implies zup(grd, porel, lam) = choice({y_ in grd: forall([c_], ord-lt(c_, lam) implies [zup(grd, porel, c_), y_] in porel)}))

## Unfold-only constructors (functoids)

`def-functoid` heads, written `name(params) := body`: the macete rewrites `name(args)` to that body, and there is no theorem — which is why these are absent from the `def-*` registry above.  `FUNCTORS.md` lists the same set split into set/structure-valued and value functoids.

TRAP worth knowing before citing one: `mac` unfolds a functoid in a GOAL, but `mac-h` **cannot** unfold it in an assumption — `def-functoid` installs no theorem for it to look up, so `mac-h` warns `unknown theorem/macete` and the driver continues with the hypothesis untouched.  A constructor whose members get read out of the CONTEXT therefore also carries a membership `iff`, stamped `definitional`, in the section below — that `iff` is a consequence of the definition here, not the definition.
