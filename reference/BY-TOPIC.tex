\subsection{Differentiation rules}

\subsubsection*{\texttt{derivative-unique}}
\emph{for every f, a, l, m, if is-diff-at(f, a, l) and is-diff-at(f, a, m), then l equals m}
{\small\[
\forall f.\; \forall a.\; \forall l.\; \forall m.\; ((\operatorname{is\text{-}diff\text{-}at}(f, a, l) \wedge \operatorname{is\text{-}diff\text{-}at}(f, a, m)) \Rightarrow (l = m))

\subsubsection*{\texttt{diff-implies-continuous}}

\emph{for every f, a, l, is-diff-at(f, a, l) if and only if f is in fun(rr, rr) and a is in rr and l is in rr and little-o-at(vnb-lambda(x, f(x) - f(a) - l * (x - a)), a)}
{\small\[
