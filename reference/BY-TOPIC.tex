\subsection{Differentiation rules}

\subsubsection*{\texttt{derivative-unique}}
\emph{for every f, a, l, m, if f is differentiable at a, with derivative l and f is differentiable at a, with derivative m, then l equals m}
{\small\[
\forall f.\; \forall a.\; \forall l.\; \forall m.\; ((\operatorname{is\text{-}diff\text{-}at}(f, a, l) \wedge \operatorname{is\text{-}diff\text{-}at}(f, a, m)) \Rightarrow (l = m))
\]}
\noindent\textbf{Machine-checked} (trust: well-known). Skolemize the two Caratheodory factors phi, psi from the two differentiability hypotheses; off the point they agree by cancelling (x-a) =/= 0, and continuity at a forces phi(a)=psi(a), i.e. L=M.
\par\smallskip\noindent\emph{Rests on 5 facts:} \texttt{eq-sym, eq-trans, rr-sub-ne-zero, rr-cancel-mul-right, cont-agree-off-pt}.

\subsubsection*{\texttt{diff-implies-continuous}}
\emph{for every f, a, l, if f is differentiable at a, with derivative l, then f is continuous at a}
{\small\[
\forall f.\; \forall a.\; \forall l.\; (\operatorname{is\text{-}diff\text{-}at}(f, a, l) \Rightarrow \operatorname{is\text{-}continuous\text{-}at}(\operatorname{rr\text{-}ms}, \operatorname{rr\text{-}ms}, f, a))
\]}
\noindent\textbf{Machine-checked} (trust: well-known). f coincides pointwise with the map x |-> f(a) + phi(x)*(x-a), which the continuity algebra makes continuous at a (a constant plus phi times x-a); continuity then transfers to f.
\par\smallskip\noindent\emph{Rests on 7 facts:} \texttt{const-continuous-at, identity-continuous-at, sub-continuous-at, product-continuous-at, sum-continuous-at, eq-sym, cont-transfer-ptwise-eq}.

\emph{for every c, a, if c is in rr and a is in rr, then vnb-lambda(x, c) is differentiable at a, with derivative 0}

\forall c.\; \forall a.\; (((c \in \mathbb{R}) \wedge (a \in \mathbb{R})) \Rightarrow \operatorname{is\text{-}diff\text{-}at}((\lambda x.\; c), a, 0))

\noindent\textbf{Machine-checked} (trust: well-known). Witness phi := 0: the difference f(x)-f(a) = 0 = 0*(x-a), and the zero map is continuous with value 0.

\subsubsection*{\texttt{deriv-sum}}
\emph{for every f, g, a, l, m, if f is differentiable at a, with derivative l and g is differentiable at a, with derivative m, then vnb-lambda(x, f(x) + g(x)) is differentiable at a, with derivative l + m}
{\small\[
\forall f.\; \forall g.\; \forall a.\; \forall l.\; \forall m.\; ((\operatorname{is\text{-}diff\text{-}at}(f, a, l) \wedge \operatorname{is\text{-}diff\text{-}at}(g, a, m)) \Rightarrow \operatorname{is\text{-}diff\text{-}at}((\lambda x.\; (f(x) + g(x))), a, (l + m)))

\subsubsection*{\texttt{deriv-product}}
\emph{for every f, g, a, l, m, if f is differentiable at a, with derivative l and g is differentiable at a, with derivative m, then vnb-lambda(x, f(x) * g(x)) is differentiable at a, with derivative l * g(a) + f(a) * m}
{\small\[
\forall f.\; \forall g.\; \forall a.\; \forall l.\; \forall m.\; ((\operatorname{is\text{-}diff\text{-}at}(f, a, l) \wedge \operatorname{is\text{-}diff\text{-}at}(g, a, m)) \Rightarrow \operatorname{is\text{-}diff\text{-}at}((\lambda x.\; (f(x) \cdot g(x))), a, ((l \cdot g(a)) + (f(a) \cdot m))))

\noindent\textbf{Asserted} [warrant: reference]. calculus.pdf Prop 2.6.  (fg)(x)-(fg)(a) = [phi\_f(x)g(x) + f(a)phi\_g(x)](x-a)

   value L*g(a)+f(a)*M.

\subsubsection*{\texttt{deriv-chain}}
\emph{for every f, g, a, l, m, if f is differentiable at a, with derivative l, then if g is differentiable at f(a), with derivative m, then compose(g, f) is differentiable at a, with derivative m * l}
{\small\[
\forall f.\; \forall g.\; \forall a.\; \forall l.\; \forall m.\; (\operatorname{is\text{-}diff\text{-}at}(f, a, l) \Rightarrow (\operatorname{is\text{-}diff\text{-}at}(g, f(a), m) \Rightarrow \operatorname{is\text{-}diff\text{-}at}(\operatorname{compose}(g, f), a, (m \cdot l))))

\noindent\textbf{Asserted} [warrant: reference]. calculus.pdf Prop 2.8 (chain rule), Caratheodory form: the factor of g o f is
   (phi\_g o f)*phi\_f, continuous at a with value g'(f(a))*f'(a) = M*L.

\subsection{Little-o calculus}

\subsubsection*{\texttt{diff-iff-little-o}}
\emph{for every f, a, l, f is differentiable at a, with derivative l if and only if f is in fun(rr, rr) and a is in rr and l is in rr and vnb-lambda(x, f(x) - f(a) - l * (x - a)) is little-o at a}
{\small\[
\forall f.\; \forall a.\; \forall l.\; (\operatorname{is\text{-}diff\text{-}at}(f, a, l) \Leftrightarrow ((f \in (\mathbb{R} \to \mathbb{R})) \wedge ((a \in \mathbb{R}) \wedge ((l \in \mathbb{R}) \wedge \operatorname{little\text{-}o\text{-}at}((\lambda x.\; \operatorname{\text{-}}(\operatorname{\text{-}}(f(x), f(a)), (l \cdot \operatorname{\text{-}}(x, a)))), a)))))
\]}
\noindent\textbf{Asserted} [warrant: reference]. calculus.pdf eq (12): f differentiable at a with derivative L iff the
   increment x|->f(x)-f(a)-L(x-a) is o(x-a) at a.  Forward: eps = phi-L for phi
   the Caratheodory factor; backward: phi = eps+L.

\subsubsection*{\texttt{little-o-sum}}
\emph{for every g, h, a, if g is little-o at a, then if h is little-o at a, then vnb-lambda(x, g(x) + h(x)) is little-o at a}
{\small\[
\forall g.\; \forall h.\; \forall a.\; (\operatorname{little\text{-}o\text{-}at}(g, a) \Rightarrow (\operatorname{little\text{-}o\text{-}at}(h, a) \Rightarrow \operatorname{little\text{-}o\text{-}at}((\lambda x.\; (g(x) + h(x))), a)))
\]}
\noindent\textbf{Asserted} [warrant: well-known]. Sum of two o(x-a) is o(x-a): add the witnesses eps\_g + eps\_h (continuous at
   a, value 0), and (g+h)(x) = (eps\_g(x)+eps\_h(x))(x-a).

\subsubsection*{\texttt{little-o-scalar}}
\emph{for every c, g, a, if c is in rr, then if g is little-o at a, then vnb-lambda(x, c * g(x)) is little-o at a}
{\small\[
\forall c.\; \forall g.\; \forall a.\; ((c \in \mathbb{R}) \Rightarrow (\operatorname{little\text{-}o\text{-}at}(g, a) \Rightarrow \operatorname{little\text{-}o\text{-}at}((\lambda x.\; (c \cdot g(x))), a)))

\subsubsection*{\texttt{extreme-value-max}}
\emph{for every f, a, b, if f is in fun(rr, rr) and a is in rr and b is in rr and a is at most b, then if for every x in ccint(a, b), f is continuous at x, then there is some c in ccint(a, b) such that for every x in ccint(a, b), f(x) is at most f(c)}
{\small\[
\forall f.\; \forall a.\; \forall b.\; (((f \in (\mathbb{R} \to \mathbb{R})) \wedge ((a \in \mathbb{R}) \wedge ((b \in \mathbb{R}) \wedge (a \leq b)))) \Rightarrow (\forall x \in \operatorname{ccint}(a, b).\; \operatorname{is\text{-}continuous\text{-}at}(\operatorname{rr\text{-}ms}, \operatorname{rr\text{-}ms}, f, x) \Rightarrow \exists c \in \operatorname{ccint}(a, b).\; \forall x \in \operatorname{ccint}(a, b).\; (f(x) \leq f(c))))
\]}
\noindent\textbf{Asserted} [warrant: reference]. Extreme Value Theorem (calculus.pdf Ch 2.5, used in Rolle's lemma): a
   continuous f on the compact [a,b] attains its supremum.  Sequentially: a
   maximizing sequence in [a,b] has a convergent subsequence (seq-compactness of
   [a,b]); its limit c is in [a,b] and f(c) is the max by sequential continuity.
