\begin{proposition}[mat-equiv-refl]
Suppose $a \text{ s.t. } \operatorname{is\text{-}ring}(a)$. Let $m, n, c \in \operatorname{mat}(m, n, \operatorname{carr}(a))$. Then
\begin{equation*}
\operatorname{mat\text{-}equiv}(a, m, n, c, c)
\end{equation*}

\end{proposition}

\begin{proof}
\begin{itemize}[leftmargin=2.8em, itemsep=4pt, parsep=1pt]
  \item[\textbf{4}] By $\operatorname{identmat\text{-}left\text{-}identity}$:
\begin{equation*}
(\operatorname{matmul}(a, \operatorname{identmat}(a, m), c) = c)
\end{equation*}

  \item[\textbf{5}] By $\operatorname{identmat\text{-}right\text{-}identity}$:
\begin{equation*}
(\operatorname{matmul}(a, c, \operatorname{identmat}(a, n)) = c)
\end{equation*}

  \item[\textbf{6}] By $\operatorname{identmat\text{-}invertible}$:
\begin{equation*}
\operatorname{is\text{-}invertible\text{-}mat}(a, m, \operatorname{identmat}(a, m))
\end{equation*}

  \item[\textbf{7}] By $\operatorname{identmat\text{-}invertible}$:
\begin{equation*}
\operatorname{is\text{-}invertible\text{-}mat}(a, n, \operatorname{identmat}(a, n))
\end{equation*}

  \item[\textbf{8}] By $\operatorname{mat\text{-}equiv}$, reduce to
\begin{equation*}
\begin{array}{@{}l@{}} \exists u. \\ \quad \operatorname{is\text{-}invertible\text{-}mat}(a, m, u) \wedge \\ \quad \exists v. \\ \quad \quad \operatorname{is\text{-}invertible\text{-}mat}(a, n, v) \wedge \\ \quad \quad (c = \operatorname{matmul}(a, \operatorname{matmul}(a, u, c), v)) \end{array}
\end{equation*}

  \item[\textbf{9}] Take $u :=$
\begin{equation*}
\operatorname{identmat}(a, m)
\end{equation*}

  \item[\textbf{10--11}] Take $w :=$
\begin{equation*}
\operatorname{identmat}(a, n)
\end{equation*}

  \item[\textbf{12--13}] Substitute the established equation.
  \item[\textbf{14--15}] Substitute the established equation.  \emph{(closes)}
\end{itemize}
\end{proof}
