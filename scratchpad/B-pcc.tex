\begin{proposition}[pivot-clears-col]
Suppose $a \text{ s.t. } \operatorname{is\text{-}euclidean\text{-}ring}(a)$. Let $m \text{ s.t. } (1 \in \operatorname{interval}(1, m))$. Suppose $n \text{ s.t. } (1 \in \operatorname{interval}(1, n))$. Let $p \in \operatorname{mat}(m, n, \operatorname{carr}(a))$. If $\neg (\operatorname{entry}(p, 1, 1) = \operatorname{zero}(a))$, then if $\forall c.\; (\operatorname{mat\text{-}equiv}(a, m, n, p, c) \Rightarrow \forall \mathit{ii}.\; \forall \mathit{jj}.\; ((\mathit{ii} \in \operatorname{interval}(1, m)) \Rightarrow ((\mathit{jj} \in \operatorname{interval}(1, n)) \Rightarrow (\neg (\operatorname{entry}(c, \mathit{ii}, \mathit{jj}) = \operatorname{zero}(a)) \Rightarrow (\operatorname{gauge}(a)(\operatorname{entry}(p, 1, 1)) \leq \operatorname{gauge}(a)(\operatorname{entry}(c, \mathit{ii}, \mathit{jj})))))))$ then Suppose $j \in \operatorname{interval}(1, n)$, if $\neg (1 = j)$ then 
\begin{equation*}
\begin{array}{@{}l@{}} \exists q. \\ \quad \operatorname{mat\text{-}equiv}(a, m, n, p, q) \wedge \\ \quad (\operatorname{entry}(q, 1, 1) = \operatorname{entry}(p, 1, 1)) \wedge \\ \quad (\operatorname{entry}(q, 1, j) = \operatorname{zero}(a)) \wedge \\ \quad \forall \mathit{ii}. \forall c. \\ \quad \quad (\mathit{ii} \in \operatorname{interval}(1, m)) \Rightarrow  \\ \quad \quad \quad (c \in \operatorname{interval}(1, n)) \Rightarrow  \\ \quad \quad \quad \quad \neg (c = j) \Rightarrow  \\ \quad \quad \quad \quad \quad (\operatorname{entry}(q, \mathit{ii}, c) = \operatorname{entry}(p, \mathit{ii}, c)) \end{array}
\end{equation*}

\end{proposition}
