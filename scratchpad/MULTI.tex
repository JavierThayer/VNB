\begin{proposition}[mat-equiv-refl]
Suppose $a \text{ s.t. } \operatorname{is\text{-}ring}(a)$. Let $m, n, c \in \operatorname{mat}(m, n, \operatorname{carr}(a))$. Then
\begin{equation*}
\operatorname{mat\text{-}equiv}(a, m, n, c, c)
\end{equation*}

\end{proposition}

\begin{proof}
\begin{itemize}[leftmargin=2.8em, itemsep=4pt, parsep=1pt]
  \item[\textbf{4}] By $\operatorname{identmat\text{-}left\text{-}identity}$:
\begin{equation*}
(\operatorname{matmul}(a, \operatorname{identmat}(a, m), c) = c)
\end{equation*}

  \item[\textbf{5}] By $\operatorname{identmat\text{-}right\text{-}identity}$:
\begin{equation*}
(\operatorname{matmul}(a, c, \operatorname{identmat}(a, n)) = c)
\end{equation*}

  \item[\textbf{6}] By $\operatorname{identmat\text{-}invertible}$:
\begin{equation*}
\operatorname{is\text{-}invertible\text{-}mat}(a, m, \operatorname{identmat}(a, m))
\end{equation*}

  \item[\textbf{7}] By $\operatorname{identmat\text{-}invertible}$:
\begin{equation*}
\operatorname{is\text{-}invertible\text{-}mat}(a, n, \operatorname{identmat}(a, n))
\end{equation*}

  \item[\textbf{8}] By $\operatorname{mat\text{-}equiv}$, reduce to
\begin{equation*}
\begin{array}{@{}l@{}} \exists u. \\ \quad \operatorname{is\text{-}invertible\text{-}mat}(a, m, u) \wedge \\ \quad \exists v. \\ \quad \quad \operatorname{is\text{-}invertible\text{-}mat}(a, n, v) \wedge \\ \quad \quad (c = \operatorname{matmul}(a, \operatorname{matmul}(a, u, c), v)) \end{array}
\end{equation*}

  \item[\textbf{9}] Take $u :=$
\begin{equation*}
\operatorname{identmat}(a, m)
\end{equation*}

  \item[\textbf{10--11}] Take $w :=$
\begin{equation*}
\operatorname{identmat}(a, n)
\end{equation*}

  \item[\textbf{12--13}] Substitute the established equation.
  \item[\textbf{14--15}] Substitute the established equation.  \emph{(closes)}
\end{itemize}
\end{proof}

\begin{proposition}[bordered-eq-border]

\begin{equation*}
(c = \operatorname{border}(a, \operatorname{entry}(c, 1, 1), \operatorname{submat}(c, p, q), p, q))

  \item[\textbf{52}] Substitute the established equation.

\end{equation*}

  \item[\textbf{80}] Substitute the established equation.
  \item[\textbf{81}] By $\operatorname{succ\text{-}nn\text{-}minus\text{-}1}$:
\begin{equation*}
((\operatorname{nn\text{-}minus}(\mathit{col}, 1) + 1) = \mathit{col})
\end{equation*}

  \item[\textbf{82}] Substitute the established equation.
  \item[\textbf{83--84}] \emph{Establish in-carrier / typing side-conditions:}

  \item[\textbf{85--86}] By $\operatorname{matrix\text{-}entry\text{-}extensionality}$:

\begin{proposition}[pivot-clears-col]
Suppose $a \text{ s.t. } \operatorname{is\text{-}euclidean\text{-}ring}(a)$. Let $m \text{ s.t. } (1 \in \operatorname{interval}(1, m))$. Suppose $n \text{ s.t. } (1 \in \operatorname{interval}(1, n))$. Let $p \in \operatorname{mat}(m, n, \operatorname{carr}(a))$. If $\neg (\operatorname{entry}(p, 1, 1) = \operatorname{zero}(a))$, then if $\forall c.\; (\operatorname{mat\text{-}equiv}(a, m, n, p, c) \Rightarrow \forall \mathit{ii}.\; \forall \mathit{jj}.\; ((\mathit{ii} \in \operatorname{interval}(1, m)) \Rightarrow ((\mathit{jj} \in \operatorname{interval}(1, n)) \Rightarrow (\neg (\operatorname{entry}(c, \mathit{ii}, \mathit{jj}) = \operatorname{zero}(a)) \Rightarrow (\operatorname{gauge}(a)(\operatorname{entry}(p, 1, 1)) \leq \operatorname{gauge}(a)(\operatorname{entry}(c, \mathit{ii}, \mathit{jj})))))))$ then Suppose $j \in \operatorname{interval}(1, n)$, if $\neg (1 = j)$ then 
\begin{equation*}
\begin{array}{@{}l@{}} \exists q. \\ \quad \operatorname{mat\text{-}equiv}(a, m, n, p, q) \wedge \\ \quad (\operatorname{entry}(q, 1, 1) = \operatorname{entry}(p, 1, 1)) \wedge \\ \quad (\operatorname{entry}(q, 1, j) = \operatorname{zero}(a)) \wedge \\ \quad \forall \mathit{ii}. \forall c. \\ \quad \quad (\mathit{ii} \in \operatorname{interval}(1, m)) \Rightarrow  \\ \quad \quad \quad (c \in \operatorname{interval}(1, n)) \Rightarrow  \\ \quad \quad \quad \quad \neg (c = j) \Rightarrow  \\ \quad \quad \quad \quad \quad (\operatorname{entry}(q, \mathit{ii}, c) = \operatorname{entry}(p, \mathit{ii}, c)) \end{array}
\end{equation*}

\end{proposition}
