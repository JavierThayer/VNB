\section*{Proof of \texttt{compact-complete}}
\textbf{Claim.}
\begin{align*}
  \forall s.\; (\mathsf{is{-}compact}(s) \Rightarrow \mathsf{is{-}complete}(s))
\end{align*}
\bigskip

\subsection*{Tactics used}
\begin{itemize}
\item \texttt{sp} -- start the proof: install the claim as the initial goal
\item \texttt{di} -- direct inference: break the goal at its top connective -- move an implication's hypothesis into the assumptions, split a conjunction, or introduce a universally quantified variable
\item \texttt{mac} -- macete: rewrite the goal using a named theorem or definition
\item \texttt{mac-h} -- macete in a hypothesis: unfold a defined predicate inside an assumption
\item \texttt{ai} -- antecedent inference: decompose a structured assumption, named by its number or its formula
\item \texttt{ass} -- assert: close the goal directly from the assumptions
\end{itemize}

\subsection*{Notation}
\begin{itemize}
\item $\Rightarrow$ -- implies
\item $\in$ -- set membership
\item $(A \to B)$ -- the set of functions from $A$ to $B$
\item $\mathbb{N}$ -- the natural numbers
\item $\wedge$ -- and
\item $\subseteq$ -- subset
\item $|x|$ -- the cardinality (number of elements) of the set $x$
\item \texttt{is-compact}$(\ldots)$ -- a defined term; see the Definitions reference (DEFINITIONS.md) for its full definition
\item \texttt{is-complete}$(\ldots)$ -- a defined term; see the Definitions reference (DEFINITIONS.md) for its full definition
\item \texttt{cluster-point}$(\ldots)$ -- a defined term; see the Definitions reference (DEFINITIONS.md) for its full definition
\item \texttt{is-cauchy-seq}$(\ldots)$ -- a defined term; see the Definitions reference (DEFINITIONS.md) for its full definition
\item \texttt{converges}$(\ldots)$ -- a defined term; see the Definitions reference (DEFINITIONS.md) for its full definition
\item \texttt{is-open-cover}$(\ldots)$ -- a defined term; see the Definitions reference (DEFINITIONS.md) for its full definition
\end{itemize}
