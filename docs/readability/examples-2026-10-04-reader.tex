\begin{proposition}[binomial-theorem]
Suppose $n \in \mathbb{N}$. Let $r \text{ be s.t. } \operatorname{is\text{-}commutative\text{-}ring}(r)$. Suppose $x \in \operatorname{carr}(r)$ and $y \in \operatorname{carr}(r)$. Then
\begin{equation*}
\begin{array}{@{}l@{}} \operatorname{ring\text{-}power}(r, \operatorname{add}(r)(x, y), n) \\ =\; \operatorname{sum}(r, \operatorname{comb\text{-}kk}(r, x, y, n), \operatorname{succ}(n)) \end{array}
\end{equation*}


\noindent\emph{(Here $\operatorname{comb\text{-}kk}(r, x, y, n)$ is the internal representation of the binomial coefficient term; it is used throughout the proof.)}

\end{proposition}

\begin{proof}
\begin{itemize}[leftmargin=2.8em, itemsep=4pt, parsep=1pt]
  \item[\textbf{1}] By induction on $n$.
  \item[]\textbf{Base case} ($n = 0$):
  \item[\textbf{2--5}] \emph{Establish in-carrier / typing side-conditions:}
\begin{equation*}
\begin{array}{@{}l@{}}
(0 \in \mathbb{N})
\end{array}
\end{equation*}

  \item[\textbf{6}] By $\operatorname{commutative\text{-}ring\text{-}is\text{-}ring}$:
\begin{equation*}
\operatorname{is\text{-}ring}(r)
\end{equation*}

  \item[\textbf{7}] \emph{Establish in-carrier / typing side-conditions:}
\begin{equation*}
\begin{array}{@{}l@{}}
(\operatorname{one}(r) \in \operatorname{carr}(r))
\end{array}
\end{equation*}

  \item[\textbf{8}] By $\operatorname{ring\text{-}power\text{-}zero}$, reduce to
\begin{equation*}
(\operatorname{one}(r) = \operatorname{sum}(r, \operatorname{comb\text{-}kk}(r, x, y, 0), (0 + 1)))
\end{equation*}

  \item[\textbf{9}] By $\operatorname{sum\text{-}succ}$, reduce to
\begin{equation*}
\begin{array}{@{}l@{}} \operatorname{one}(r) \\ =\; \operatorname{add}(r)(\operatorname{sum}(r, \operatorname{comb\text{-}kk}(r, x, y, 0), 0), \operatorname{comb\text{-}kk}(r, x, y, 0)(0)) \end{array}
\end{equation*}

  \item[\textbf{10}] By $\operatorname{sum\text{-}zero}$, reduce to
\begin{equation*}
\begin{array}{@{}l@{}} \operatorname{one}(r) \\ =\; \operatorname{add}(r)(\operatorname{zero}(r), \operatorname{comb\text{-}kk}(r, x, y, 0)(0)) \end{array}
\end{equation*}

  \item[\textbf{11}] By $\operatorname{comb\text{-}kk\text{-}0\text{-}0}$:
\begin{equation*}
(\operatorname{comb\text{-}kk}(r, x, y, 0)(0) = \operatorname{one}(r))
\end{equation*}

  \item[\textbf{12}] Substitute the established equation.
  \item[\textbf{13--14}] By $\operatorname{ring\text{-}add\text{-}left\text{-}id}$, reduce to
\begin{equation*}
(\operatorname{one}(r) = \operatorname{one}(r))
\end{equation*}
  \emph{(closes)}
  \item[]\textbf{Inductive step} ($n \to n + 1$):
  \item[\textbf{15--24}] \emph{Establish in-carrier / typing side-conditions:}
\begin{equation*}
\begin{array}{@{}l@{}}
(\operatorname{comb\text{-}kk}(r, x, y, n) \in (\mathbb{Z} \to \operatorname{carr}(r))) \\
((n + 1) \in \mathbb{N}) \\
((0 - 1) \in \mathbb{Z}) \\
(n \in \mathbb{Z}) \\
((n + 1) \in \mathbb{Z})
\end{array}
\end{equation*}

  \item[\textbf{25}] By $\operatorname{commutative\text{-}ring\text{-}is\text{-}ring}$:
\begin{equation*}
\operatorname{is\text{-}ring}(r)
\end{equation*}

  \item[\textbf{26--27}] \emph{Establish in-carrier / typing side-conditions:}
\begin{equation*}
\begin{array}{@{}l@{}}
(\operatorname{add}(r)(x, y) \in \operatorname{carr}(r)) \\
\begin{array}{@{}l@{}} \operatorname{sum}(r, \operatorname{comb\text{-}kk}(r, x, y, n), (n + 1)) \\ \in\; \operatorname{carr}(r) \end{array}
\end{array}
\end{equation*}

  \item[\textbf{28}] By $\operatorname{bt\text{-}neg1\text{-}neg}$:
\begin{equation*}
((0 - 1) < 0)
\end{equation*}

  \item[\textbf{29}] By $\operatorname{bt\text{-}lt\text{-}succ}$:
\begin{equation*}
(n < (n + 1))
\end{equation*}

  \item[\textbf{30}] By $\operatorname{ring\text{-}power\text{-}succ}$, reduce to
\begin{equation*}
\begin{array}{@{}l@{}} \operatorname{mul}(r)(\operatorname{ring\text{-}power}(r, \operatorname{add}(r)(x, y), n), \operatorname{add}(r)(x, y)) \\ =\; \operatorname{sum}(r, \operatorname{comb\text{-}kk}(r, x, y, (n + 1)), ((n + 1) + 1)) \end{array}
\end{equation*}

  \item[\textbf{31--33}] Instantiate the hypothesis:
\begin{equation*}
\begin{array}{@{}l@{}} \operatorname{ring\text{-}power}(r, \operatorname{add}(r)(x, y), n) \\ =\; \operatorname{sum}(r, \operatorname{comb\text{-}kk}(r, x, y, n), (n + 1)) \end{array}
\end{equation*}

  \item[\textbf{34}] Substitute the established equation.
  \item[\textbf{35}] By $\operatorname{comb\text{-}kk\text{-}succ}$, reduce to
\begin{equation*}
\begin{array}{@{}l@{}} \operatorname{mul}(r)(\operatorname{sum}(r, \operatorname{comb\text{-}kk}(r, x, y, n), (n + 1)), \operatorname{add}(r)(x, y)) \\ =\; \operatorname{sum}(r, \\ \quad (\lambda k \in \mathbb{Z}.\; \\ \quad \quad \operatorname{add}(r)(\operatorname{mul}(r)(x, \operatorname{comb\text{-}kk}(r, x, y, n)((k - 1))), \\ \quad \quad \quad \operatorname{mul}(r)(y, \operatorname{comb\text{-}kk}(r, x, y, n)(k)))), \\ \quad ((n + 1) + 1)) \end{array}
\end{equation*}

  \item[\textbf{36}] By $\operatorname{sum\text{-}expansion}$:
\begin{equation*}
\begin{array}{@{}l@{}} \operatorname{sum}(r, \\ \quad (\lambda k \in \mathbb{Z}.\; \\ \quad \quad \operatorname{add}(r)(\operatorname{mul}(r)(x, \operatorname{comb\text{-}kk}(r, x, y, n)((k - 1))), \\ \quad \quad \quad \operatorname{mul}(r)(y, \operatorname{comb\text{-}kk}(r, x, y, n)(k)))), \\ \quad ((n + 1) + 1)) \\ =\; \operatorname{add}(r)(\operatorname{add}(r)(\operatorname{mul}(r)(\operatorname{add}(r)(x, y), \operatorname{sum}(r, \operatorname{comb\text{-}kk}(r, x, y, n), (n + 1))), \\ \quad \quad \operatorname{mul}(r)(x, \operatorname{comb\text{-}kk}(r, x, y, n)((0 - 1)))), \\ \quad \operatorname{mul}(r)(y, \operatorname{comb\text{-}kk}(r, x, y, n)((n + 1)))) \end{array}
\end{equation*}

  \item[\textbf{37}] Substitute the established equation.
  \item[\textbf{38}] By $\operatorname{comb\text{-}kk\text{-}null}$:
\begin{equation*}
(\operatorname{comb\text{-}kk}(r, x, y, n)((0 - 1)) = \operatorname{zero}(r))
\end{equation*}

  \item[\textbf{39}] Substitute the established equation.
  \item[\textbf{40}] By $\operatorname{comb\text{-}kk\text{-}above}$:
\begin{equation*}
(\operatorname{comb\text{-}kk}(r, x, y, n)((n + 1)) = \operatorname{zero}(r))
\end{equation*}

  \item[\textbf{41--42}] Substitute the established equation.  \emph{(closes)}
\end{itemize}
\end{proof}

\begin{proposition}[recip-succ-small]
Suppose $d \text{ s.t. } \operatorname{pos\text{-}rr}(d)$. There is $\operatorname{n\_} \in \mathbb{N}$ such that for every $\operatorname{m\_} \in \mathbb{N}$, if $(\operatorname{n\_} \leq \operatorname{m\_})$ then 
\begin{equation*}
(\operatorname{recip}(+(\operatorname{m\_}, 1)) \leq d)
\end{equation*}

\end{proposition}

\begin{proof}
\begin{itemize}[leftmargin=2.8em, itemsep=4pt, parsep=1pt]
  \item[\textbf{3}] By $\operatorname{rr\text{-}recip\text{-}succ\text{-}converges\text{-}to\text{-}zero}$:
\begin{equation*}
\operatorname{converges\text{-}to}(\operatorname{rr\text{-}ms}, (\lambda k \in \mathbb{N}.\; \operatorname{recip}((k + 1))), 0)
\end{equation*}

  \item[\textbf{4}] By $\operatorname{mac\text{-}h}$.
  \item[\textbf{5--8}] Instantiate the hypothesis:
\begin{equation*}
\begin{array}{@{}l@{}} \exists n \in \mathbb{N}. \forall \operatorname{n\_} \in \mathbb{N}. \\ \quad (n \leq \operatorname{n\_}) \Rightarrow  \\ \quad \quad \operatorname{dist}(\operatorname{rr\text{-}ms})((\lambda k \in \mathbb{N}.\; \operatorname{recip}((k + 1)))(\operatorname{n\_}), 0) \\ \quad \quad \leq\; d \end{array}
\end{equation*}

  \item[\textbf{9--11}] Take $\operatorname{n\_} :=$ $n$.
  \item[\textbf{12--15}] Instantiate the hypothesis:
\begin{equation*}
\begin{array}{@{}l@{}} \operatorname{dist}(\operatorname{rr\text{-}ms})((\lambda k \in \mathbb{N}.\; \operatorname{recip}((k + 1)))(\operatorname{m\_}), 0) \\ \leq\; d \end{array}
\end{equation*}

  \item[\textbf{16}] By $\operatorname{lam\text{-}b\text{-}h}$.
  \item[\textbf{17--19}] \emph{Establish in-carrier / typing side-conditions:}
\begin{equation*}
\begin{array}{@{}l@{}}
(1 \in \mathbb{R}) \\
(0 \in \mathbb{R}) \\
(\operatorname{m\_} \in \mathbb{R})
\end{array}
\end{equation*}

  \item[\textbf{20}] By $\operatorname{nn\text{-}zero\text{-}le}$:
\begin{equation*}
(0 \leq \operatorname{m\_})
\end{equation*}

  \item[\textbf{21}] Introduce the auxiliary claim
\begin{equation*}
((\operatorname{m\_} + 1) \in \mathbb{R})
\end{equation*}

  \item[\textbf{22--25}] Introduce the auxiliary claim
\begin{equation*}
((\operatorname{m\_} \in \mathbb{R}) \wedge (1 \in \mathbb{R}))
\end{equation*}
  \emph{(side goal holds by assumption)}
  \item[\textbf{26--27}] By $\operatorname{rr\text{-}add\text{-}closed}$:
\begin{equation*}
\begin{array}{@{}l@{}} ((\operatorname{m\_} \in \mathbb{R}) \wedge (1 \in \mathbb{R})) \Rightarrow  \\ \quad ((\operatorname{m\_} + 1) \in \mathbb{R}) \end{array}
\end{equation*}
  \emph{(holds by assumption)}
  \item[\textbf{28--29}] Introduce the auxiliary claim
\begin{equation*}
(0 < (\operatorname{m\_} + 1))
\end{equation*}
  \emph{(side goal discharged)}
  \item[\textbf{30}] By $\operatorname{rr\text{-}pos\text{-}ne\text{-}zero}$:
\begin{equation*}
\neg ((\operatorname{m\_} + 1) = 0)
\end{equation*}

  \item[\textbf{31}] Introduce the auxiliary claim
\begin{equation*}
(((\operatorname{m\_} + 1) \in \mathbb{R}) \wedge \neg ((\operatorname{m\_} + 1) = 0))
\end{equation*}

  \item[\textbf{32--33}] By $\operatorname{rr\text{-}recip\text{-}closed}$:
\begin{equation*}
(\operatorname{recip}((\operatorname{m\_} + 1)) \in \mathbb{R})
\end{equation*}

  \item[\textbf{34}] By $\operatorname{rr\text{-}recip\text{-}pos}$:
\begin{equation*}
(0 < \operatorname{recip}((\operatorname{m\_} + 1)))
\end{equation*}

  \item[\textbf{35}] By $\operatorname{mac\text{-}h}$.
  \item[\textbf{36}] By $\operatorname{mac\text{-}h}$.
  \item[\textbf{37--39}] Introduce the auxiliary claim
\begin{equation*}
((\operatorname{recip}((\operatorname{m\_} + 1)) - 0) \in \mathbb{R})
\end{equation*}

  \item[\textbf{40--43}] Introduce the auxiliary claim
\begin{equation*}
((\operatorname{recip}((\operatorname{m\_} + 1)) \in \mathbb{R}) \wedge (0 \in \mathbb{R}))
\end{equation*}
  \emph{(side goal holds by assumption)}
  \item[\textbf{44--45}] By $\operatorname{rr\text{-}sub\text{-}in\text{-}rr}$:
\begin{equation*}
\begin{array}{@{}l@{}} (0 \in \mathbb{R}) \Rightarrow  \\ \quad ((\operatorname{recip}((\operatorname{m\_} + 1)) - 0) \in \mathbb{R}) \end{array}
\end{equation*}
  \emph{(holds by assumption)}
  \item[\textbf{46--48}] By $\operatorname{mac\text{-}h}$.  \emph{(closes)}
\end{itemize}
\end{proof}

\begin{proposition}[interval-widen]
Suppose $b \in \mathbb{N}$, $c \in \mathbb{N}$, $a$ and $i \in \operatorname{interval}(a, b)$. If $(b \leq c)$, then 
\begin{equation*}
(i \in \operatorname{interval}(a, c))
\end{equation*}

\end{proposition}

\begin{proof}
\begin{itemize}[leftmargin=2.8em, itemsep=4pt, parsep=1pt]
  \item[\textbf{3}] \emph{Establish in-carrier / typing side-conditions:}
\begin{equation*}
\begin{array}{@{}l@{}}
(i \in \mathbb{N})
\end{array}
\end{equation*}

  \item[\textbf{4}] By $\operatorname{interval\text{-}lo}$:
\begin{equation*}
(a \leq i)
\end{equation*}

  \item[\textbf{5}] By $\operatorname{interval\text{-}hi}$:
\begin{equation*}
(i \leq b)
\end{equation*}

  \item[\textbf{6}] By $\operatorname{nn\text{-}le\text{-}trans\text{-}guarded}$:
\begin{equation*}
(i \leq c)
\end{equation*}

  \item[\textbf{7--8}] \emph{Establish in-carrier / typing side-conditions:}
\begin{equation*}
\begin{array}{@{}l@{}}
(i \in \operatorname{interval}(a, c))
\end{array}
\end{equation*}
  \emph{(side goal holds by assumption)}
\end{itemize}
\end{proof}

\begin{proposition}[card-singleton]
Suppose $x \in \mathbf{Set}$. Then
\begin{equation*}
(|\{x\}| = \operatorname{succ}(0))
\end{equation*}

\end{proposition}

\begin{proof}
\begin{itemize}[leftmargin=2.8em, itemsep=4pt, parsep=1pt]
  \item[\textbf{2}] Introduce the auxiliary claim
\begin{equation*}
((\emptyset \cup \{x\}) = \{x\})
\end{equation*}

  \item[\textbf{3}] Backchain through $\operatorname{class\text{-}extensionality}$; it remains to show
\begin{equation*}
(|\{x\}| = (0 + 1))
\end{equation*}

  \item[\textbf{4--5}] By $\operatorname{union\text{-}membership}$:
\begin{equation*}
((x \in (\emptyset \cup \{x\})) \Leftrightarrow ((x \in \emptyset) \vee (x \in \{x\})))
\end{equation*}

  \item[\textbf{6}] By $\operatorname{empty\text{-}set\text{-}has\text{-}no\text{-}members}$:
\begin{equation*}
\neg (x \in \emptyset)
\end{equation*}

  \item[\textbf{7--15}] By $\operatorname{prop}$.

  \item[\textbf{17}] By $\operatorname{empty\text{-}set\text{-}has\text{-}no\text{-}members}$:

\neg (x \in \emptyset)
\end{equation*}

  \item[\textbf{18}] By $\operatorname{card\text{-}empty}$:

(|\emptyset| = 0)
\end{equation*}

  \item[\textbf{19}] \emph{Establish in-carrier / typing side-conditions:}
\begin{equation*}
\begin{array}{@{}l@{}}
(0 \in \mathbb{N})

  \item[\textbf{28}] By $\operatorname{ord\text{-}succ\text{-}nn}$:
\begin{equation*}

\end{equation*}

  \item[\textbf{29}] Substitute the established equation.
  \item[\textbf{30}] Substitute the established equation.
  \item[\textbf{31--32}] Substitute the established equation.  \emph{(holds by assumption)}
