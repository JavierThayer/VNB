\begin{equation*}
\begin{array}{@{}l@{}} \operatorname{series\text{-}partial\text{-}sum}(\operatorname{h\_}, \operatorname{succ}(\operatorname{n\_})) \\ =\; \cdot(\cdot(\operatorname{n\_}, \operatorname{x\_}), -(1, \operatorname{x\_})) \end{array}

\end{proposition}
