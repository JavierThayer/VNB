\section*{Proof of \texttt{cc-is-metric-space}}
\textbf{Claim.}
\begin{align*}
  \operatorname{is\text{-}metric\text{-}space}(\operatorname{cc\text{-}ms})
\end{align*}
\bigskip

\subsection*{Tactics used}
\begin{itemize}
\item \texttt{sp} -- start the proof: install the claim as the initial goal
\item \texttt{mac} -- macete: rewrite the goal using a named theorem or definition
\item \texttt{di} -- direct inference: break the goal at its top connective -- move an implication's hypothesis into the assumptions, split a conjunction, or introduce a universally quantified variable
\item \texttt{ass} -- assert: close the goal directly from the assumptions
\item \texttt{mac-h} -- macete in a hypothesis: unfold a defined predicate inside an assumption
\item \texttt{cut} -- cut: introduce an intermediate lemma, discharged as a side goal
\item \texttt{subst} -- substitute: rewrite the goal using an equality assumption
\end{itemize}

\subsection*{Notation}
\begin{itemize}
\item $\Rightarrow$ -- implies
\item $\wedge$ -- and
\item $\in$ -- set membership
\item $\mathbb{C}$ -- the complex numbers
\item $\mathbb{R}$ -- the real numbers
\item $\langle \ldots \rangle$ -- a finite tuple
\item $\times$ -- Cartesian product
\item $(CARR \to B)$ -- the set of functions from $A$ to $B$
\item \texttt{is-metric}$(\ldots)$ -- a defined term; see the Definitions reference (DEFINITIONS.md) for its full definition
\end{itemize}
