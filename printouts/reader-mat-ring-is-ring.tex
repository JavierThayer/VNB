\begin{proposition}[mat-ring-is-ring]
Suppose $a \text{ s.t. } \operatorname{is\text{-}ring}(a)$. Let $n \in \mathbb{N}$. Then
\begin{equation*}
\operatorname{is\text{-}ring}(\operatorname{mat\text{-}ring}(a, n))
\end{equation*}

\end{proposition}
