\begin{proposition}[rolle]
Suppose $h, a$. Let $b \text{ s.t. } ((h \in (\mathbb{R} \to \mathbb{R})) \wedge ((a \in \mathbb{R}) \wedge ((b \in \mathbb{R}) \wedge (a < b))))$. If $\forall x \in \operatorname{ccint}(a, b).\; \operatorname{is\text{-}continuous\text{-}at}(\operatorname{rr\text{-}ms}, \operatorname{rr\text{-}ms}, h, x)$, then if $\forall x.\; (((a < x) \wedge (x < b)) \Rightarrow \exists l.\; \operatorname{is\text{-}diff\text{-}at}(h, x, l))$ then if $(h(a) = h(b))$ then 
\begin{equation*}
\begin{array}{@{}l@{}} \exists \theta. \\ \quad ((a < \theta) \wedge ((\theta < b) \wedge \operatorname{is\text{-}diff\text{-}at}(h, \theta, 0))) \end{array}
\end{equation*}

\end{proposition}
